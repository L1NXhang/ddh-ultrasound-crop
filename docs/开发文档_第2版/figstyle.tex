% ---------- 基本样式 ----------
\tikzset{

  % 判定节点
  dec/.style={base, diamond, aspect=1.7, draw=rust, line width=0.55pt, fill=rustf,

  % 强调节点
  hot/.style={base, rectangle, rounded corners=1.6pt, draw=rust, line width=0.7pt,

  tensor/.style={base, rectangle, draw=ink2, line width=0.4pt, fill=white,

  dasharr/.style={-{Stealth[length=1.9mm,width=1.3mm]}, line width=0.55pt,

  % 标签（边上文字）
  lbl/.style={base, font=\figfontS, text=ink2, inner sep=1.1pt,

  badgeR/.style={badge, draw=rust, text=rust},
